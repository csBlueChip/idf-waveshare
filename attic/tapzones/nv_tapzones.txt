== src/nv.c checkers
/* Tap calibration grid "x:y" (ROT 0), x * y 1..16. */
static const char *nvChkTapXy(const char *in, int min, int max, char *out)
{
	int  x, y;
	char extra;
	if (sscanf(in, "%d:%d%c", &x, &y, &extra) != 2 || x < 1 || y < 1 || x * y > 16) {
		return "must be x:y with x * y 1..16";
	}
	sprintf(out, "%d:%d", x, y);
	return NULL;
}

/* Tap zone centre "x:y:z" (gx/a : gy/a : gz/a, IMU TAP n CAL; "x:y" = z 0);
 * 0:0:0 = none. */
static const char *nvChkTapZone(const char *in, int min, int max, char *out)
{
	float x, y, z = 0.0f;
	char  extra;
	int   n = sscanf(in, "%f:%f:%f%c", &x, &y, &z, &extra);
	if ((n != 2 && n != 3) || !isfinite(x) || !isfinite(y) || !isfinite(z) || fabsf(x) > 9999.0f ||
	    fabsf(y) > 9999.0f || fabsf(z) > 9999.0f) {
		return "must be x:y:z (numbers; 0:0:0 = not calibrated)";
	}
	sprintf(out, "%.3f:%.3f:%.3f", x, y, z);
	if (strcmp(out, "0.000:0.000:0.000") == 0) {
		strcpy(out, "0:0:0");
	}
	return NULL;
}
