# ---------- NV / RAD (settings are backed up and restored) ----------
NV EXPORT selftest_backup              # SD:/sys/selftest_backup.nvram

NV SET rad.pwr 7.3                     # stored as 7.25

NV SET btn.l 10                        # ERR btn.l must be 50..5000

RAD GET

RAD SET ip 192.168.4.255               # ERR rad.ip is not a host address for rad.mask
RAD SET ssid ok pwd bad                # ERR ... (nothing stored)
RAD GET ssid
NV RESET scr.bl
NV GET scr.bl                          # scr.bl=100
NV IMPORT missing                      # ERR file not found
NV RESET                               # all defaults
NV IMPORT selftest_backup              # back to what you had
NV GET
